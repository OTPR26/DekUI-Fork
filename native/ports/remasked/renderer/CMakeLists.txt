cmake_minimum_required(VERSION 3.30)
project(ReMaskedNativeRenderer LANGUAGES C CXX)
if(NOT APPLE)
  message(FATAL_ERROR "This entry point qualifies the Apple Vulkan/Metal backend")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(REMASKED_MOLTENVK_ROOT "" CACHE PATH "Verified MoltenVK 1.4.2 package root")
set(REMASKED_MOLTENVK_SLICE "" CACHE STRING "XCFramework platform slice")
set(molten "${REMASKED_MOLTENVK_ROOT}/MoltenVK/static/MoltenVK.xcframework/${REMASKED_MOLTENVK_SLICE}/libMoltenVK.a")
if(NOT EXISTS "${molten}")
  message(FATAL_ERROR "Supply a verified static MoltenVK package and platform slice")
endif()
set(Vulkan_LIBRARY "${molten}" CACHE FILEPATH "Static MoltenVK" FORCE)
set(Vulkan_INCLUDE_DIR "${REMASKED_MOLTENVK_ROOT}/MoltenVK/include" CACHE PATH "MoltenVK headers" FORCE)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
include(FetchContent)
set(NRI_STATIC_LIBRARY ON CACHE BOOL "" FORCE)
set(NRI_ENABLE_VK_SUPPORT ON CACHE BOOL "" FORCE)
foreach(feature NONE NVTX WGPU)
  set(NRI_ENABLE_${feature}_SUPPORT OFF CACHE BOOL "" FORCE)
endforeach()
set(NRI_ENABLE_NIS_SDK OFF CACHE BOOL "" FORCE)
set(NRI_ENABLE_NGX_SDK OFF CACHE BOOL "" FORCE)
set(NRI_ENABLE_FFX_SDK OFF CACHE BOOL "" FORCE)
set(NRI_ENABLE_VALIDATION_SUPPORT ON CACHE BOOL "" FORCE)
FetchContent_Declare(nri
  URL https://github.com/NVIDIA-RTX/NRI/archive/4b485316463969f182db15e67aad2aec2f40a3d7.tar.gz
  PATCH_COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/../tools/patch_nri_static_vulkan.py" <SOURCE_DIR>)
FetchContent_MakeAvailable(nri)
target_compile_definitions(NRI_Shared PRIVATE REMASKED_STATIC_MOLTENVK=1)
target_compile_definitions(NRI PRIVATE REMASKED_STATIC_MOLTENVK=1)
target_link_libraries(NRI PUBLIC "${molten}" "-framework Metal" "-framework Foundation"
  "-framework QuartzCore" "-framework IOSurface")
if(CMAKE_SYSTEM_NAME STREQUAL "Darwin")
  target_link_libraries(NRI PUBLIC "-framework IOKit")
endif()
add_executable(remasked_apple_nri_probe ../../android/native/nri_probe.cpp)
target_link_libraries(remasked_apple_nri_probe PRIVATE NRI)
add_library(remasked_renderer_probe STATIC renderer_probe.cpp)
target_link_libraries(remasked_renderer_probe PUBLIC NRI)
