cmake_minimum_required(VERSION 3.26)
project(ReMaskedNativePrototype LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
get_filename_component(TRIAEVUM_ROOT "../.." ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
set(REMASKED_JSON_INCLUDE "" CACHE PATH "Pinned nlohmann 3.12.0 include directory")
if(NOT EXISTS "${REMASKED_JSON_INCLUDE}/nlohmann/json.hpp")
  include(FetchContent)
  set(JSON_BuildTests OFF CACHE BOOL "" FORCE)
  FetchContent_Declare(remasked_json
    URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
    URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
  FetchContent_MakeAvailable(remasked_json)
  set(REMASKED_JSON_INCLUDE "${remasked_json_SOURCE_DIR}/include")
endif()
add_library(remasked_mobile_core STATIC
  shared/remasked_core.cpp
  "${TRIAEVUM_ROOT}/titles/mm3d/dekui/settings.cpp"
  "${TRIAEVUM_ROOT}/runtime/triaevum_module/src/input_service.cpp")
target_include_directories(remasked_mobile_core PUBLIC shared PRIVATE
  "${TRIAEVUM_ROOT}/titles/mm3d/dekui"
  "${TRIAEVUM_ROOT}/runtime/triaevum_module/include" "${REMASKED_JSON_INCLUDE}")
target_compile_options(remasked_mobile_core PRIVATE
  "-ffile-prefix-map=${TRIAEVUM_ROOT}=." "-fmacro-prefix-map=${TRIAEVUM_ROOT}=.")
if(ANDROID)
  add_library(remasked_mobile SHARED android/app/src/main/cpp/bridge.cpp)
  target_link_libraries(remasked_mobile PRIVATE remasked_mobile_core)
  target_link_options(remasked_mobile PRIVATE -Wl,-z,max-page-size=16384)
  set(REMASKED_TITLE_BUILD "" CACHE PATH "Private matching Android arm64 static-title build")
  if(REMASKED_TITLE_BUILD)
    foreach(name remasked_static_title_host remasked_static_title triaevum_title_whole_aot_support)
      set(archive "${REMASKED_TITLE_BUILD}/lib${name}.a")
      if(NOT EXISTS "${archive}")
        message(FATAL_ERROR "Missing private static title library: ${name}")
      endif()
      target_link_libraries(remasked_mobile PRIVATE "${archive}")
    endforeach()
    target_compile_definitions(remasked_mobile PRIVATE REMASKED_NATIVE_TITLE=1)
  endif()
  set(REMASKED_NRI_BUILD "" CACHE PATH "Matching Android NRI probe build")
  if(REMASKED_NRI_BUILD)
    target_sources(remasked_mobile PRIVATE renderer/renderer_probe.cpp)
    target_include_directories(remasked_mobile PRIVATE "${REMASKED_NRI_BUILD}/_deps/nri-src/Include")
    foreach(name NRI NRI_VK NRI_Validation NRI_Shared)
      set(archive "${REMASKED_NRI_BUILD}/_deps/nri-build/lib${name}.a")
      if(NOT EXISTS "${archive}")
        message(FATAL_ERROR "Missing native renderer library: ${name}")
      endif()
      target_link_libraries(remasked_mobile PRIVATE "${archive}")
    endforeach()
    target_link_libraries(remasked_mobile PRIVATE log dl)
    target_compile_definitions(remasked_mobile PRIVATE REMASKED_NATIVE_NRI=1 NRI_STATIC_LIBRARY=1)
  endif()
endif()
if(NOT CMAKE_CROSSCOMPILING)
  enable_testing()
  add_executable(remasked_mobile_tests tests/core_tests.cpp)
  target_include_directories(remasked_mobile_tests PRIVATE "${REMASKED_JSON_INCLUDE}")
  target_link_libraries(remasked_mobile_tests PRIVATE remasked_mobile_core)
  add_test(NAME mobile_preservation_and_input COMMAND remasked_mobile_tests
    "${CMAKE_CURRENT_SOURCE_DIR}/resources/presentation.json")
endif()
